cmake_minimum_required(VERSION 3.16)
project(godot_extended_images LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

if(NOT DEFINED GODOTCPP_PLATFORM)
    if(WIN32)
        set(GODOTCPP_PLATFORM "windows")
    elseif(APPLE)
        set(GODOTCPP_PLATFORM "macos")
    else()
        set(GODOTCPP_PLATFORM "linux")
    endif()
endif()

if(NOT DEFINED AVIF_GDEXTENSION_TARGET)
    string(TOLOWER "${CMAKE_BUILD_TYPE}" _bt_lower)
    if(_bt_lower STREQUAL "debug" OR _bt_lower STREQUAL "relwithdebinfo")
        set(AVIF_GDEXTENSION_TARGET "template_debug")
    else()
        set(AVIF_GDEXTENSION_TARGET "template_release")
    endif()
endif()

if(NOT DEFINED AVIF_GDEXTENSION_ARCH)
    set(AVIF_GDEXTENSION_ARCH "${CMAKE_SYSTEM_PROCESSOR}")
    if(AVIF_GDEXTENSION_ARCH STREQUAL "AMD64" OR AVIF_GDEXTENSION_ARCH STREQUAL "x86")
        set(AVIF_GDEXTENSION_ARCH "x86_64")
    elseif(AVIF_GDEXTENSION_ARCH STREQUAL "aarch64")
        set(AVIF_GDEXTENSION_ARCH "arm64")
    endif()
endif()

message(STATUS "godot-extended-images: platform=${GODOTCPP_PLATFORM} target=${AVIF_GDEXTENSION_TARGET} arch=${AVIF_GDEXTENSION_ARCH}")

# 1. godot-cpp (master = Godot 4.7 API)
set(GODOTCPP_PLATFORM ${GODOTCPP_PLATFORM} CACHE STRING "" FORCE)
if(NOT DEFINED GODOTCPP_API_VERSION)
    set(GODOTCPP_API_VERSION "4.7" CACHE STRING "" FORCE)
endif()
add_subdirectory(thirdparty/godot-cpp)

# 2. libavif + dav1d
set(AVIF_CODEC_DAV1D "LOCAL" CACHE STRING "" FORCE)
set(AVIF_CODEC_AOM "OFF" CACHE STRING "" FORCE)
set(AVIF_CODEC_LIBGAV1 "OFF" CACHE STRING "" FORCE)
set(AVIF_CODEC_RAV1E "OFF" CACHE STRING "" FORCE)
set(AVIF_CODEC_SVT "OFF" CACHE STRING "" FORCE)
set(AVIF_CODEC_AVM "OFF" CACHE STRING "" FORCE)
set(AVIF_BUILD_APPS OFF CACHE BOOL "" FORCE)
set(AVIF_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(AVIF_LIBYUV "OFF" CACHE STRING "" FORCE)
set(AVIF_LIBSHARPYUV "OFF" CACHE STRING "" FORCE)
set(AVIF_JPEG "OFF" CACHE STRING "" FORCE)
set(AVIF_ZLIBPNG "OFF" CACHE STRING "" FORCE)
option(AVIF_DAV1D_DISABLE_ASM "Build dav1d without hand-written assembly (no NASM required)" ON)
if(AVIF_DAV1D_DISABLE_ASM)
    set(AVIF_DAV1D_ENABLE_ASM "false" CACHE STRING "" FORCE)
else()
    set(AVIF_DAV1D_ENABLE_ASM "true" CACHE STRING "" FORCE)
endif()
add_subdirectory(thirdparty/libavif)

# 3. libjxl
set(JPEGXL_ENABLE_TOOLS OFF CACHE BOOL "" FORCE)
set(JPEGXL_ENABLE_MANPAGES OFF CACHE BOOL "" FORCE)
set(JPEGXL_ENABLE_BENCHMARK OFF CACHE BOOL "" FORCE)
set(JPEGXL_ENABLE_EXAMPLES OFF CACHE BOOL "" FORCE)
set(JPEGXL_ENABLE_SJPEG OFF CACHE BOOL "" FORCE)
set(JPEGXL_ENABLE_OPENEXR OFF CACHE BOOL "" FORCE)
set(JPEGXL_ENABLE_DOXYGEN OFF CACHE BOOL "" FORCE)
set(JPEGXL_ENABLE_PLUGINS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(JPEGXL_ENABLE_SKCMS ON CACHE BOOL "" FORCE)
set(JPEGXL_ENABLE_VIEWERS OFF CACHE BOOL "" FORCE)
add_subdirectory(thirdparty/libjxl)

# 4. libde265 (HEVC decoder for libheif)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(ENABLE_SDL OFF CACHE BOOL "" FORCE)
set(ENABLE_DECODER ON CACHE BOOL "" FORCE)
set(ENABLE_ENCODER OFF CACHE BOOL "" FORCE)
add_subdirectory(thirdparty/libde265)

# 5. libheif (HEIC/HEIF container)
set(WITH_LIBDE265 ON CACHE BOOL "" FORCE)
set(WITH_X265 OFF CACHE BOOL "" FORCE)
set(WITH_AOM_DECODER OFF CACHE BOOL "" FORCE)
set(WITH_AOM_ENCODER OFF CACHE BOOL "" FORCE)
set(WITH_DAV1D OFF CACHE BOOL "" FORCE)
set(WITH_OPENH264_DECODER OFF CACHE BOOL "" FORCE)
set(WITH_X264 OFF CACHE BOOL "" FORCE)
set(WITH_UNCOMPRESSED OFF CACHE BOOL "" FORCE)
set(ENABLE_PLUGIN_LOADING OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WITH_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WITH_GDK_PIXBUF OFF CACHE BOOL "" FORCE)
# Point FindLIBDE265.cmake at the built libde265
set(LIBDE265_INCLUDE_DIR "${CMAKE_SOURCE_DIR}/thirdparty/libde265" CACHE PATH "" FORCE)
set(LIBDE265_LIBRARY "de265" CACHE STRING "" FORCE)
add_subdirectory(thirdparty/libheif)

# libde265's de265-version.h is generated by CMake into the build dir —
# add it to heif's include path so the build can find it.
target_include_directories(heif PRIVATE "${CMAKE_BINARY_DIR}/thirdparty/libde265")

# libde265 declares functions with __declspec(dllimport) on Windows unless
# LIBDE265_STATIC_BUILD is defined. Since we link statically, both the heif
# library and our extension need this define.
target_compile_definitions(heif PRIVATE LIBDE265_STATIC_BUILD)

# 6. Our GDExtension shared library
set(EXT_SRC
    src/register_types.cpp
    src/image_decoders.cpp
    src/image_loader_avif.cpp
    src/image_loader_jxl.cpp
    src/image_loader_heic.cpp
    src/image_loader_qoi.cpp
    src/resource_format_loader_extended.cpp
    src/editor_preview_generator_extended.cpp
    src/extended_images_editor_plugin.cpp
    src/extended_image_info.cpp
)

set(_ext_suffix "")
if(GODOTCPP_PLATFORM STREQUAL "windows")
    set(_ext_suffix "dll")
elseif(GODOTCPP_PLATFORM STREQUAL "macos")
    set(_ext_suffix "dylib")
else()
    set(_ext_suffix "so")
endif()

set(_output_name "libgodot_extended_images.${GODOTCPP_PLATFORM}.${AVIF_GDEXTENSION_TARGET}.${AVIF_GDEXTENSION_ARCH}")
set(_output_dir "${CMAKE_SOURCE_DIR}/bin/${GODOTCPP_PLATFORM}")

add_library(godot_extended_images SHARED ${EXT_SRC})

set_target_properties(godot_extended_images PROPERTIES
    OUTPUT_NAME "${_output_name}"
    PREFIX ""
    SUFFIX ".${_ext_suffix}"
    RUNTIME_OUTPUT_DIRECTORY "${_output_dir}"
    LIBRARY_OUTPUT_DIRECTORY "${_output_dir}"
    ARCHIVE_OUTPUT_DIRECTORY "${_output_dir}"
)

target_compile_definitions(godot_extended_images PRIVATE
    JXL_STATIC_DEFINE
    HWY_STATIC_DEFINE
    LIBDE265_STATIC_BUILD
)

target_include_directories(godot_extended_images PRIVATE
    "${CMAKE_SOURCE_DIR}/src"
    "${CMAKE_SOURCE_DIR}/thirdparty/libavif/include"
    "${CMAKE_SOURCE_DIR}/thirdparty/libjxl/lib/include"
    "${CMAKE_BINARY_DIR}/thirdparty/libjxl/lib/include"
    "${CMAKE_SOURCE_DIR}/thirdparty/libjxl/third_party/highway"
    "${CMAKE_SOURCE_DIR}/thirdparty/libjxl/third_party/brotli/c/include"
    "${CMAKE_SOURCE_DIR}/thirdparty/libheif/libheif/api"
    "${CMAKE_BINARY_DIR}/thirdparty/libheif"
    "${CMAKE_SOURCE_DIR}/thirdparty/libde265"
    "${CMAKE_BINARY_DIR}/thirdparty/libde265"
    "${CMAKE_SOURCE_DIR}/thirdparty/qoi"
)

# Link all libraries
set(_jxl_build_lib_dir "${CMAKE_BINARY_DIR}/thirdparty/libjxl/lib")
set(_jxl_brotli_lib_dir "${CMAKE_BINARY_DIR}/thirdparty/libjxl/third_party/brotli")
set(_jxl_highway_lib_dir "${CMAKE_BINARY_DIR}/thirdparty/libjxl/third_party/highway")
set(_lib_suf ".a")

target_link_libraries(godot_extended_images PRIVATE
    godot-cpp
    avif
    heif
    de265
    ${_jxl_build_lib_dir}/libjxl${_lib_suf}
    ${_jxl_build_lib_dir}/libjxl_cms${_lib_suf}
    ${_jxl_build_lib_dir}/libjxl_dec-internal${_lib_suf}
    ${_jxl_brotli_lib_dir}/libbrotlidec${_lib_suf}
    ${_jxl_brotli_lib_dir}/libbrotlienc${_lib_suf}
    ${_jxl_brotli_lib_dir}/libbrotlicommon${_lib_suf}
    ${_jxl_highway_lib_dir}/libhwy${_lib_suf}
)
# NOTE: brotlicommon must come AFTER brotlidec/brotlienc: GNU ld processes
# archives left-to-right and would otherwise discard it before dec/enc pull
# in BrotliDefaultAllocFunc & friends (broke loading in a real Godot editor).

if(GODOTCPP_PLATFORM STREQUAL "linux")
    target_link_libraries(godot_extended_images PRIVATE pthread)
endif()
